# Timing constraints used for final implementation
## Smart Medicine Reminder and Missed Alert Controller
## Constraints (XDC style – final GDS / board pin mapping)

## Clock
set_property PACKAGE_PIN W5 [get_ports clk]
set_property IOSTANDARD LVCMOS33 [get_ports clk]
create_clock -add -name sys_clk -period 100.00 -waveform {0 50} [get_ports clk]

## Reset (active low)
set_property PACKAGE_PIN V17 [get_ports rst_n]
set_property IOSTANDARD LVCMOS33 [get_ports rst_n]

## Inputs
set_property PACKAGE_PIN T18 [get_ports tick_1hz]
set_property PACKAGE_PIN U18 [get_ports taken]
set_property PACKAGE_PIN W19 [get_ports load_sched]
set_property IOSTANDARD LVCMOS33 [get_ports {tick_1hz taken load_sched}]

## set_hour[4:0]
set_property PACKAGE_PIN U17 [get_ports {set_hour[0]}]
set_property PACKAGE_PIN W16 [get_ports {set_hour[1]}]
set_property PACKAGE_PIN V16 [get_ports {set_hour[2]}]
set_property PACKAGE_PIN V15 [get_ports {set_hour[3]}]
set_property PACKAGE_PIN W14 [get_ports {set_hour[4]}]
set_property IOSTANDARD LVCMOS33 [get_ports {set_hour[*]}]

## set_min[5:0]
set_property PACKAGE_PIN W13 [get_ports {set_min[0]}]
set_property PACKAGE_PIN V2  [get_ports {set_min[1]}]
set_property PACKAGE_PIN T3  [get_ports {set_min[2]}]
set_property PACKAGE_PIN T2  [get_ports {set_min[3]}]
set_property PACKAGE_PIN R3  [get_ports {set_min[4]}]
set_property PACKAGE_PIN W2  [get_ports {set_min[5]}]
set_property IOSTANDARD LVCMOS33 [get_ports {set_min[*]}]

## miss_window[5:0]
set_property PACKAGE_PIN R2  [get_ports {miss_window[0]}]
set_property PACKAGE_PIN T1  [get_ports {miss_window[1]}]
set_property PACKAGE_PIN U1  [get_ports {miss_window[2]}]
set_property PACKAGE_PIN W7  [get_ports {miss_window[3]}]
set_property PACKAGE_PIN W6  [get_ports {miss_window[4]}]
set_property PACKAGE_PIN V7  [get_ports {miss_window[5]}]
set_property IOSTANDARD LVCMOS33 [get_ports {miss_window[*]}]

## Outputs
set_property PACKAGE_PIN U16 [get_ports reminder]
set_property PACKAGE_PIN E19 [get_ports missed_alert]
set_property IOSTANDARD LVCMOS33 [get_ports {reminder missed_alert}]

## cur_hour / cur_min (debug LEDs – optional)
set_property PACKAGE_PIN U19 [get_ports {cur_hour[0]}]
set_property PACKAGE_PIN V19 [get_ports {cur_hour[1]}]
set_property PACKAGE_PIN W18 [get_ports {cur_hour[2]}]
set_property PACKAGE_PIN U15 [get_ports {cur_hour[3]}]
set_property PACKAGE_PIN U14 [get_ports {cur_hour[4]}]
set_property IOSTANDARD LVCMOS33 [get_ports {cur_hour[*]}]
